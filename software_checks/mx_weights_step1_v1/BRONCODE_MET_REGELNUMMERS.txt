0067: class qfn(torch.autograd.Function):
0068:     @staticmethod
0069:     def forward(ctx, input, k):
0070:         n = float(2**k - 1)
0071:         out = torch.round(input * n) / n
0072:         return out
0073: 
0074:     @staticmethod
0075:     def backward(ctx, grad_output):
0076:         grad_input = grad_output.clone()
0077:         return grad_input, None

0080: class weight_quantize_fn(nn.Module):
0081:     def __init__(self, bit_list):
0082:         super(weight_quantize_fn, self).__init__()
0083:         self.bit_list = bit_list
0084:         self.wbit = self.bit_list[-1]
0085:         assert self.wbit <= 8 or self.wbit == 32
0086: 
0087:     def forward(self, x):
0088:         if self.wbit == 32:
0089:             E = torch.mean(torch.abs(x)).detach()
0090:             weight = torch.tanh(x)
0091:             weight = weight / torch.max(torch.abs(weight))
0092:             weight_q = weight * E
0093:         else:
0094:             E = torch.mean(torch.abs(x)).detach()
0095:             weight = torch.tanh(x)
0096:             weight = weight / 2 / torch.max(torch.abs(weight)) + 0.5
0097:             weight_q = 2 * qfn.apply(weight, self.wbit) - 1
0098:             weight_q = weight_q * E
0099:         return weight_q

0014: class Activate(nn.Module):
0015:     def __init__(self, bit_list, quantize=True):
0016:         super(Activate, self).__init__()
0017:         self.bit_list = bit_list
0018:         self.abit = self.bit_list[-1]
0019:         self.acti = nn.ReLU(inplace=True)
0020:         self.quantize = quantize
0021:         if self.quantize:
0022:             self.quan = activation_quantize_fn(bit_list=self.bit_list)
0023: 
0024:     def forward(self, x):
0025:         if self.abit == 32:
0026:             x = self.acti(x)
0027:         else:
0028:             x = torch.clamp(x, 0.0, 1.0)
0029:         if self.quantize:
0030:             x = self.quan(x)
0031:         return x

0034: class SVHNQ(nn.Module):
0035:     def __init__(self, bit_list, num_classes, expand=8):
0036:         super(SVHNQ, self).__init__()
0037:         self.bit_list = bit_list
0038:         self.wbit = self.bit_list[-1]
0039:         self.abit = self.bit_list[-1]
0040:         Conv2d = conv2d_quantize_fn(bit_list=self.bit_list)
0041:         NormLayer = batchnorm_fn(self.bit_list)
0042:         self.expand = expand
0043: 
0044:         ep = self.expand
0045:         self.layers = nn.Sequential(
0046:             nn.Conv2d(3, ep * 6, 5, padding=0, bias=True),
0047:             nn.MaxPool2d(2),
0048:             Activate(self.bit_list),
0049:             # 18
0050:             Conv2d(ep * 6, ep * 8, 3, padding=1, bias=False),
0051:             NormLayer(ep * 8),
0052:             Activate(self.bit_list),
0053:             Conv2d(ep * 8, ep * 8, 3, padding=1, bias=False),
0054:             NormLayer(ep * 8),
0055:             nn.MaxPool2d(2),
0056:             Activate(self.bit_list),
0057:             # 9
0058:             Conv2d(ep * 8, ep * 16, 3, padding=0, bias=False),
0059:             NormLayer(ep * 16),
0060:             Activate(self.bit_list),
0061:             # 7
0062:             Conv2d(ep * 16, ep * 16, 3, padding=1, bias=False),
0063:             NormLayer(ep * 16),
0064:             Activate(self.bit_list),
0065:             Conv2d(ep * 16, ep * 16, 3, padding=0, bias=False),
0066:             NormLayer(ep * 16),
0067:             Activate(self.bit_list),
0068:             # 5
0069:             nn.Dropout(0.5),
0070:             Conv2d(ep * 16, ep * 64, 5, padding=0, bias=False),
0071:             Activate(self.bit_list, quantize=False),
0072:         )
0073:         self.fc = nn.Linear(ep * 64, 10)
0074: 
0075:     def forward(self, x):
0076:         x = self.layers(x)
0077:         x = x.view([x.shape[0], -1])
0078:         x = self.fc(x)
0079:         return x

0083:     return SVHNQ(bit_list, num_classes=num_classes, expand=expand)
